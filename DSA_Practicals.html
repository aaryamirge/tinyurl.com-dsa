1. Searching Techniques

Linear Search and Binary Search

Linear search checks each element. Binary search works only on a sorted array.

#include <stdio.h>
int main() {
    int a[100], n, key, i, found = 0;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter elements: ");
    for(i=0;i<n;i++) scanf("%d",&a[i]);
    printf("Enter element to search: ");
    scanf("%d",&key);
    for(i=0;i<n;i++) {
        if(a[i] == key) { found=1; break; }
    }
    if(found) printf("Element found at position %d",i+1);
    else printf("Element not found");
    return 0;
}
  • What is searching? Finding the location of a required element.
  • Linear search? Checks elements one by one.
  • Binary search condition? Array must be sorted.
  • Linear search time complexity? O(n).
  • Binary search time complexity? O(log n).

2. Sorting Algorithms

Bubble Sort (ascending)

#include <stdio.h>
int main() {
    int a[100], n, i, j, temp;
    printf("Enter number of elements: ");
    scanf("%d",&n);
    printf("Enter elements: ");
    for(i=0;i<n;i++) scanf("%d",&a[i]);
    for(i=0;i<n-1;i++) {
        for(j=0;j<n-i-1;j++) {
            if(a[j] > a[j+1]) {
                temp=a[j]; a[j]=a[j+1]; a[j+1]=temp;
            }
        }
    }
    printf("Sorted array: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    return 0;
}
  • What is sorting? Arranging data in ascending or descending order.
  • What is Bubble Sort? Repeatedly compares adjacent elements and swaps them if needed.
  • Bubble Sort average/worst time? O(n²).
  • What is swapping? Interchanging two values.

3. Student Database Management

Uses an array of structures to store roll number, name and marks.

#include <stdio.h>
struct Student { int roll; char name[50]; float marks; };
int main() {
    struct Student s[50];
    int n,i;
    printf("Enter number of students: ");
    scanf("%d",&n);
    for(i=0;i<n;i++) {
        printf("Enter roll, name and marks: ");
        scanf("%d %49s %f",&s[i].roll,s[i].name,&s[i].marks);
    }
    printf("\\nStudent Records\\n");
    for(i=0;i<n;i++)
        printf("%d %s %.2f\\n",s[i].roll,s[i].name,s[i].marks);
    return 0;
}
  • What is a structure? A user-defined data type that groups different data types.
  • Why use an array of structures? To store records of many students.
  • What is a database? An organized collection of data.
  • What does %f read? A floating-point value.

4. Stack Using Array

Stack follows LIFO: Last In, First Out.

#include <stdio.h>
#define MAX 5
int stack[MAX], top=-1;
void push(int x) {
    if(top==MAX-1) printf("Stack Overflow\\n");
    else stack[++top]=x;
}
void pop() {
    if(top==-1) printf("Stack Underflow\\n");
    else printf("Deleted: %d\\n",stack[top--]);
}
void display() {
    int i;
    if(top==-1) printf("Stack is empty\\n");
    else for(i=top;i>=0;i--) printf("%d\\n",stack[i]);
}
int main() {
    push(10); push(20); push(30);
    display(); pop(); display();
    return 0;
}
  • What is stack? A linear data structure following LIFO.
  • What is push? Inserting an element.
  • What is pop? Removing the top element.
  • Overflow? Trying to insert into a full stack.
  • Underflow? Trying to delete from an empty stack.

5. Linked List Operations

Singly linked list: each node contains data and a pointer to the next node.

#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node *next; };
struct Node *head=NULL;
void insertEnd(int x) {
    struct Node *n=malloc(sizeof(struct Node)), *t;
    n->data=x; n->next=NULL;
    if(head==NULL) head=n;
    else { t=head; while(t->next!=NULL) t=t->next; t->next=n; }
}
void deleteBeginning() {
    struct Node *t;
    if(head==NULL) printf("List is empty\\n");
    else { t=head; head=head->next; free(t); }
}
void display() {
    struct Node *t=head;
    while(t!=NULL) { printf("%d -> ",t->data); t=t->next; }
    printf("NULL\\n");
}
int main() {
    insertEnd(10); insertEnd(20); insertEnd(30);
    display(); deleteBeginning(); display();
    return 0;
}
  • What is linked list? A collection of nodes connected using links/pointers.
  • What is a node? A data field and link field.
  • What is head? Pointer to the first node.
  • What does NULL mean? Pointer does not point to a valid node.
  • Array vs linked list? Array uses contiguous memory; linked-list nodes need not.

6. Generation of Patterns

Right triangle and inverted triangle

#include <stdio.h>
int main() {
    int i,j,n;
    printf("Enter rows: ");
    scanf("%d",&n);
    for(i=1;i<=n;i++) {
        for(j=1;j<=i;j++) printf("* ");
        printf("\\n");
    }
    printf("\\nInverted pattern:\\n");
    for(i=n;i>=1;i--) {
        for(j=1;j<=i;j++) printf("* ");
        printf("\\n");
    }
    return 0;
}
  • Which loops are used? Nested for loops.
  • What is an outer loop used for? Controls rows.
  • What is an inner loop used for? Prints symbols/columns in each row.
  • What does printf("\\n") do? Moves output to the next line.

7. Queue Using Linked List

Queue follows FIFO: First In, First Out.

#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node *next; };
struct Node *front=NULL, *rear=NULL;
void enqueue(int x) {
    struct Node *n=malloc(sizeof(struct Node));
    n->data=x; n->next=NULL;
    if(rear==NULL) front=rear=n;
    else { rear->next=n; rear=n; }
}
void dequeue() {
    struct Node *t;
    if(front==NULL) printf("Queue is empty\\n");
    else {
        t=front; printf("Deleted: %d\\n",t->data);
        front=front->next;
        if(front==NULL) rear=NULL;
        free(t);
    }
}
void display() {
    struct Node *t=front;
    while(t!=NULL) { printf("%d ",t->data); t=t->next; }
    printf("\\n");
}
int main() {
    enqueue(10); enqueue(20); enqueue(30);
    display(); dequeue(); display();
    return 0;
}
  • What is queue? A linear data structure following FIFO.
  • What is enqueue? Insertion at rear.
  • What is dequeue? Deletion from front.
  • What are front and rear? Pointers to first and last queue nodes.

8. Height and Depth of Tree

This example calculates binary-tree height in edges. Empty tree height is -1; a leaf has height 0.

#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node *left,*right; };
int max(int a,int b) { return a>b?a:b; }
int height(struct Node *root) {
    if(root==NULL) return -1;
    return 1+max(height(root->left),height(root->right));
}
int main() {
    struct Node *root=malloc(sizeof(struct Node));
    root->data=1;
    root->left=malloc(sizeof(struct Node));
    root->left->data=2;
    root->left->left=NULL; root->left->right=NULL;
    root->right=malloc(sizeof(struct Node));
    root->right->data=3;
    root->right->left=NULL; root->right->right=NULL;
    printf("Height = %d\\n",height(root));
    return 0;
}
  • What is a tree? A non-linear hierarchical data structure.
  • What is root? The topmost node.
  • What is a leaf? A node with no children.
  • What is height? Number of edges on the longest downward path to a leaf.
  • What is depth? Number of edges from root to a node.

9. BST Traversal

In a Binary Search Tree, smaller values go left and larger values go right. Inorder traversal gives sorted order.

#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node *left,*right; };
struct Node *insert(struct Node *r,int x) {
    if(r==NULL) {
        r=malloc(sizeof(struct Node));
        r->data=x; r->left=r->right=NULL;
    } else if(x<r->data) r->left=insert(r->left,x);
    else if(x>r->data) r->right=insert(r->right,x);
    return r;
}
void inorder(struct Node *r) {
    if(r) { inorder(r->left); printf("%d ",r->data); inorder(r->right); }
}
void preorder(struct Node *r) {
    if(r) { printf("%d ",r->data); preorder(r->left); preorder(r->right); }
}
void postorder(struct Node *r) {
    if(r) { postorder(r->left); postorder(r->right); printf("%d ",r->data); }
}
int main() {
    struct Node *root=NULL;
    root=insert(root,50); root=insert(root,30); root=insert(root,70);
    root=insert(root,20); root=insert(root,40);
    printf("Inorder: "); inorder(root);
    printf("\\nPreorder: "); preorder(root);
    printf("\\nPostorder: "); postorder(root);
    return 0;
}
  • What is BST? A binary tree where left values are smaller and right values are larger than the node.
  • Inorder? Left, Root, Right.
  • Preorder? Root, Left, Right.
  • Postorder? Left, Right, Root.
  • Which traversal of BST gives sorted data? Inorder.

10. Graph Traversal – BFS and DFS

Adjacency matrix representation. BFS uses a queue; DFS uses recursion (call stack).

#include <stdio.h>
int a[10][10], visited[10], n;
void DFS(int v) {
    int i;
    printf("%d ",v);
    visited[v]=1;
    for(i=0;i<n;i++)
        if(a[v][i] && !visited[i]) DFS(i);
}
void BFS(int start) {
    int q[10], front=0,rear=0,i,v;
    for(i=0;i<n;i++) visited[i]=0;
    q[rear++]=start; visited[start]=1;
    while(front<rear) {
        v=q[front++]; printf("%d ",v);
        for(i=0;i<n;i++)
            if(a[v][i] && !visited[i]) {
                q[rear++]=i; visited[i]=1;
            }
    }
}
int main() {
    int i,j,start;
    printf("Enter number of vertices: ");
    scanf("%d",&n);
    printf("Enter adjacency matrix:\\n");
    for(i=0;i<n;i++)
        for(j=0;j<n;j++) scanf("%d",&a[i][j]);
    printf("Enter starting vertex (0 to %d): ",n-1);
    scanf("%d",&start);
    for(i=0;i<n;i++) visited[i]=0;
    printf("DFS: "); DFS(start);
    printf("\\nBFS: "); BFS(start);
    return 0;
}
  • What is graph? A set of vertices connected by edges.
  • What is BFS? Breadth First Search explores level by level and uses a queue.
  • What is DFS? Depth First Search explores deeply before backtracking; it can use recursion or a stack.
  • What is adjacency matrix? A 2D array representing connections between vertices.
  • Why visited array? To avoid visiting a vertex repeatedly.
